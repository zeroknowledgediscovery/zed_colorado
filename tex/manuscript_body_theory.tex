\section{Main theorem}

The original deterministic argument can be made materially more realistic by separating a \emph{high-probability} residual-genomic bound from the stronger special case of a literal uniform bound.

\paragraph*{Notation.} Write
\begin{equation}
R_G(g;c)=\log \LR_{G\mid C}(g;c)
\end{equation}
for the residual genomic log-likelihood ratio. For a clinical likelihood-ratio threshold $\lambda>0$, define the clinical-only and clinically-augmented decisions
\begin{equation}
D_C=\ind\{\LR_C\ge \lambda\},\qquad D_{C,G}=\ind\{\LR_C\LR_{G\mid C}\ge \lambda\}.
\end{equation}

\textbf{A1$^\star$ (Uniform stochastic boundedness of residual genomic evidence).} For the sequential clinical histories $C_n=(X_1,\ldots,X_n)$, write
\begin{equation}
R_{G,n}=\log \LR_{G\mid C_n}(G;C_n).
\end{equation}
For each tolerance level $\delta\in(0,1)$ relevant to the operating regime, there exists a finite constant $b_\delta<\infty$, independent of $n$, such that
\begin{equation}
\sup_n\Pzero\bigl(|R_{G,n}|>b_\delta\bigr)\le \delta.
\label{eq:stochasticbound}
\end{equation}
Thus the residual genomic log-likelihood contribution remains tight as clinical observations accumulate. This condition allows rare large residual genomic effects; it excludes only a residual genomic scale that grows without bound with clinical-history length.

\textbf{A2 (Extended clinical-evidence tail).} Clinical history can accumulate arbitrarily strong disease evidence on rare trajectories. Formally, for arbitrarily large $K$ there exists an $n$ such that
\begin{equation}
\Pzero\bigl(\log \LR_{C_n}(C_n)\ge K\bigr)>0.
\label{eq:clinicaltailassumption}
\end{equation}
A2 is an essential-unboundedness condition, not a claim that typical controls acquire large likelihood ratios.

\begin{theorem}[High-probability boundary localization and clinical-tail dominance]
\label{thm:main}
Assume A1$^\star$.

\emph{(i) Exact factorization.} The joint disease-versus-control likelihood ratio factorizes as
\begin{equation}
\LR_{C,G}(c,g)=\LR_C(c)\LR_{G\mid C}(g;c).
\end{equation}

\emph{(ii) High-probability boundary localization.} For any decision threshold $\lambda>0$ and any $\delta\in(0,1)$ with bound $b_\delta$ from \eqref{eq:stochasticbound},
\begin{equation}
\boxed{
\Pzero\Bigl(D_{C,G}\neq D_C,\ \bigl|\log \LR_C-\log\lambda\bigr|>b_\delta\Bigr)\le \delta.
}
\label{eq:stochastic-band-result}
\end{equation}
Thus, outside a clinical log-LR band of half-width $b_\delta$ around the decision boundary, genomic information can reverse the clinical decision only on a control-side exceptional event of probability at most $\delta$.

\emph{(iii) High-probability clinical-tail dominance.} For every $n$, every $K>0$, and every $\delta$ with bound $b_\delta$ from A1$^\star$,
\begin{equation}
\boxed{
\Pzero\left(
\log\LR_{C_n}\ge K,\
\left|\frac{\log\LR_{C_n,G}}{\log\LR_{C_n}}-1\right|>\frac{b_\delta}{K}
\right)\le\delta.
}
\label{eq:taildom-prob}
\end{equation}
If A2 also holds, positive-probability clinical tail events exist for arbitrarily large $K$, while the tolerated relative contribution $b_\delta/K$ tends to zero. Thus, apart from the same low-probability residual-genomic exception, sufficiently strong clinical log-likelihood dominates the combined log-likelihood.
\end{theorem}

\begin{proof}
Part (i) is the chain rule:
\begin{align}
\LR_{C,G}(c,g)
&=\frac{p(c,g\mid Y=1)}{p(c,g\mid Y=0)}\\
&=\frac{p(c\mid Y=1)p(g\mid Y=1,c)}{p(c\mid Y=0)p(g\mid Y=0,c)}\\
&=\LR_C(c)\LR_{G\mid C}(g;c).
\end{align}

For part (ii), if both $|\log \LR_C-\log\lambda|>b_\delta$ and $|R_G|\le b_\delta$ hold, then the sign of $\log \LR_C-\log\lambda$ cannot be changed by adding $R_G$. Indeed, if $\log \LR_C-\log\lambda>b_\delta$, then
\begin{equation}
\log \LR_{C,G}-\log\lambda = \log \LR_C-\log\lambda + R_G > b_\delta-b_\delta = 0,
\end{equation}
so $D_{C,G}=D_C=1$. Similarly, if $\log \LR_C-\log\lambda<-b_\delta$, then $\log \LR_{C,G}-\log\lambda<0$ and $D_{C,G}=D_C=0$. Therefore
\begin{equation}
\Bigl\{D_{C,G}\neq D_C,\ |\log \LR_C-\log\lambda|>b_\delta\Bigr\}\subseteq \{|R_G|>b_\delta\},
\end{equation}
which yields \eqref{eq:stochastic-band-result} by A1$^\star$.

For part (iii), on the event $\{|R_{G,n}|\le b_\delta\}\cap\{\log\LR_{C_n}\ge K\}$,
\begin{equation}
\left|\frac{\log\LR_{C_n,G}}{\log\LR_{C_n}}-1\right|
=\frac{|R_{G,n}|}{\log\LR_{C_n}}
\le \frac{b_\delta}{K}.
\end{equation}
Therefore violation of the relative bound on the clinical tail implies $|R_{G,n}|>b_\delta$, whose $\Pzero$ probability is at most $\delta$ uniformly in $n$ by A1$^\star$. A2 guarantees that such clinical tail events occur at arbitrarily large $K$, making $b_\delta/K$ arbitrarily small.
\end{proof}

\begin{corollary}[Deterministic bounded-residual special case]
\label{cor:deterministic-band}
If there exist constants $0<m\le 1\le M<\infty$ such that
\begin{equation}
0<m\le \LR_{G\mid C}(g;c)\le M<\infty
\label{eq:boundedconditional}
\end{equation}
for $\Pzero$-almost every relevant $(g,c)$, then A1$^\star$ holds with
\begin{equation}
b_\delta \equiv b=\max\{\log M,-\log m\}
\end{equation}
for every $\delta>0$. Consequently, genomic information can change which side of the threshold an observation occupies only if
\begin{equation}
\boxed{\frac{\lambda}{M}\le \LR_C(c)\le \frac{\lambda}{m}.}
\label{eq:boundaryband}
\end{equation}
Equivalently,
\begin{equation}
\boxed{-\log M\le \log \LR_C(c)-\log\lambda\le -\log m.}
\label{eq:logboundaryband}
\end{equation}
Hence the deterministic decision-relevant band has fixed width $\log(M/m)$ on the clinical log-likelihood scale.
\end{corollary}

\begin{proof}
Under \eqref{eq:boundedconditional}, $R_G=\log \LR_{G\mid C}$ lies in $[\log m,\log M]$, so $|R_G|\le b$ almost surely. Applying Theorem~\ref{thm:main}(ii) with $\delta=0$ gives the exact localization claim. Writing out the two one-sided cases recovers \eqref{eq:boundaryband}--\eqref{eq:logboundaryband}.
\end{proof}

\begin{corollary}[Posterior-risk form for ZeBRA]
\label{cor:zebra}
If ZeBRA is calibrated so that $Z=\mathbb{P}(Y=1\mid C)$, let $\tau\in(0,1)$ be any ZeBRA decision threshold and define posterior-odds threshold $\kappa=\tau/(1-\tau)$. Under Corollary~\ref{cor:deterministic-band}, genomic information can change the decision only if
\begin{equation}
\boxed{\frac{\kappa}{M}\le \frac{Z}{1-Z}\le \frac{\kappa}{m}.}
\label{eq:zboundary}
\end{equation}
Equivalently,
\begin{equation}
-\log M\le \operatorname{logit}(Z)-\operatorname{logit}(\tau)\le -\log m.
\end{equation}
Thus genomic utility is localized around whichever ZeBRA operating boundary is chosen.
\end{corollary}

\begin{proof}
Equation \eqref{eq:zlr} shows that ZeBRA posterior odds differ from $\LR_C$ only by the positive constant prior odds $\pi/(1-\pi)$. Substituting into \eqref{eq:boundaryband} yields the result.
\end{proof}

\begin{proposition}[Genomic-only power bound and extreme-specificity clinical superiority]
\label{prop:genomicpower}
Assume the marginal genomic likelihood ratio
\begin{equation}
\LR_G(g)=\frac{p(g\mid Y=1)}{p(g\mid Y=0)}
\end{equation}
is bounded above by $M_G<\infty$ and A2 holds in the form that for arbitrarily large $K$ there are events with positive $\Pzero$ probability on which $\LR_C\ge K$. Then:

\emph{(i)} for any genomic-only test $\phi(G)\in[0,1]$ with false-positive rate $\alpha=\Ezero[\phi(G)]$,
\begin{equation}
\boxed{\Eone[\phi(G)]\le M_G\alpha;}
\label{eq:genomicpowerbound}
\end{equation}

\emph{(ii)} for any $K>M_G$ with $\alpha_K=\Pzero(\LR_C\ge K)>0$, the clinical likelihood-ratio rule
\begin{equation}
\phi_K(C)=\ind\{\LR_C(C)\ge K\}
\end{equation}
has
\begin{equation}
\frac{\Pone(\LR_C\ge K)}{\Pzero(\LR_C\ge K)}\ge K>M_G,
\label{eq:clinicalLRplus}
\end{equation}
so at the same FPR $\alpha_K$ its power exceeds that of every genomic-only rule. Moreover $\alpha_K\le 1/K\to0$ as $K\to\infty$.
\end{proposition}

\begin{proof}
For part (i), using the Radon--Nikodym identity under $\Pzero$,
\begin{align}
\Eone[\phi(G)]
&=\Ezero[\LR_G(G)\phi(G)]\\
&\le M_G\Ezero[\phi(G)]\\
&=M_G\alpha.
\end{align}
For part (ii), define $A_K=\{c:\LR_C(c)\ge K\}$. Then
\begin{align}
\Pone(A_K)
&=\Ezero[\LR_C(C)\ind\{A_K\}]\\
&\ge K\Pzero(A_K)=K\alpha_K,
\end{align}
which gives \eqref{eq:clinicalLRplus}. If $K>M_G$, then part (i) implies that any genomic-only test with false-positive rate $\alpha_K$ has power at most $M_G\alpha_K<K\alpha_K\le \Pone(A_K)$. Markov's inequality applied to $\Ezero[\LR_C]=1$ yields $\alpha_K\le 1/K$.
\end{proof}

\begin{corollary}[Any genomic model inherits the marginal bound]
Let $S=f(G)$ be any measurable genomic predictor, including a polygenic risk score, boosted tree, random forest, neural network, or other deterministic transformation of the measured genomic state. If $\LR_G$ is bounded above by $M_G$, then the likelihood ratio of $S$ is also bounded above by $M_G$. Therefore no deterministic compression of the same genomic information can evade Proposition~\ref{prop:genomicpower}.
\end{corollary}

\begin{proof}
For a score value $s$, the score likelihood ratio is the conditional expectation
\begin{equation}
\LR_S(s)=\mathbb{E}_0[\LR_G(G)\mid S=s].
\end{equation}
A conditional expectation of a random variable bounded above by $M_G$ is itself bounded above by $M_G$, $\Pzero$-almost surely on the score support.
\end{proof}

\begin{proposition}[Local decision gain can coexist with negligible global AUC change]
\label{prop:auc-local}
Let $S(C)$ be a clinical score and $S'(C,G)$ an augmented score. Suppose $S'=S$ except on a measurable region $B$ of the clinical--genomic state space. Then
\begin{align}
\left|\operatorname{AUC}(S')-\operatorname{AUC}(S)\right|
&\le \Pone(B)+\Pzero(B)-\Pone(B)\Pzero(B)\\
&\le \Pone(B)+\Pzero(B).
\label{eq:auc-local-bound}
\end{align}
Hence an augmented rule can produce a substantial change in decisions within a small boundary region while producing an arbitrarily small change in global AUC as the case and control mass of that region tends to zero.
\end{proposition}

\begin{proof}
AUC is the probability that a randomly drawn case receives a higher score than a randomly drawn control, with the usual half-credit convention for ties. If neither member of a case--control pair lies in $B$, then $S'=S$ for both members and the pairwise ranking contribution is unchanged. Therefore the AUC contribution can differ only when at least one member of the pair lies in $B$. The probability of that event is
\begin{multline}
1-(1-\Pone(B))(1-\Pzero(B))\\
=\Pone(B)+\Pzero(B)-\Pone(B)\Pzero(B).
\end{multline}
which bounds the maximum possible AUC change.
\end{proof}
